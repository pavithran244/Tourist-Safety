def calculate_distance_meters(
    latitude1,
    longitude1,
    latitude2,
    longitude2
):
    earth_radius = 6371000

    latitude1_radians = radians(latitude1)
    latitude2_radians = radians(latitude2)

    difference_latitude = radians(
        latitude2 - latitude1
    )

    difference_longitude = radians(
        longitude2 - longitude1
    )

    a = (
        sin(difference_latitude / 2) ** 2
        +
        cos(latitude1_radians)
        *
        cos(latitude2_radians)
        *
        sin(difference_longitude / 2) ** 2
    )

    c = 2 * atan2(
        sqrt(a),
        sqrt(1 - a)
    )

    distance = earth_radius * c

    return distance

def check_geofence(latitude, longitude):

    distance = calculate_distance_meters(
        latitude,
        longitude,
        GEOFENCE_CENTER_LATITUDE,
        GEOFENCE_CENTER_LONGITUDE
    )

    inside = distance <= GEOFENCE_RADIUS_METERS

    if inside:
        status = "SAFE"
    else:
        status = "OUTSIDE_GEOFENCE"

    return {
        "inside": inside,
        "status": status,
        "distance_meters": round(distance, 2),
        "radius_meters": GEOFENCE_RADIUS_METERS,
        "center": {
            "latitude": GEOFENCE_CENTER_LATITUDE,
            "longitude": GEOFENCE_CENTER_LONGITUDE
        }
    }

@app.post("/location")
def receive_location(data: LocationData):

    geofence = check_geofence(
        data.latitude,
        data.longitude
    )

    timestamp = datetime.now().isoformat()

    tourist = {
        "tourist_id": data.tourist_id,
        "latitude": data.latitude,
        "longitude": data.longitude,
        "accuracy": data.accuracy,
        "status": geofence["status"],
        "inside_geofence": geofence["inside"],
        "distance_from_geofence_meters": geofence["distance_meters"],
        "last_updated": timestamp
    }

    tourists[data.tourist_id] = tourist


    # --------------------------------------------------------
    # LOCATION HISTORY
    # --------------------------------------------------------

    if data.tourist_id not in location_history:
        location_history[data.tourist_id] = []

    location_history[data.tourist_id].append(
        {
            "latitude": data.latitude,
            "longitude": data.longitude,
            "accuracy": data.accuracy,
            "status": geofence["status"],
            "time": timestamp
        }
    )


    # Keep latest 100 locations
    if len(location_history[data.tourist_id]) > 100:
        location_history[data.tourist_id] = (
            location_history[data.tourist_id][-100:]
        )


    return {
        "success": True,
        "message": "Location received",
        "tourist": tourist,
        "geofence": geofence
    }

@app.post("/sos")
def receive_sos(data: SOSData):

    timestamp = datetime.now().isoformat()

    geofence = check_geofence(
        data.latitude,
        data.longitude
    )

    alert = {
        "id": len(alerts) + 1,
        "tourist_id": data.tourist_id,
        "latitude": data.latitude,
        "longitude": data.longitude,
        "accuracy": data.accuracy,
        "message": data.message,
        "time": timestamp,
        "status": "ACTIVE",
        "inside_geofence": geofence["inside"],
        "distance_from_geofence_meters": geofence["distance_meters"]
    }

    alerts.append(alert)


    if data.tourist_id in tourists:

        tourists[data.tourist_id]["status"] = "SOS"

        tourists[data.tourist_id]["last_updated"] = timestamp


    return {
        "success": True,
        "message": "SOS received by authority",
        "alert": alert
    }

  ] = useState(false);


  const [
    status,
    setStatus
  ] = useState("SAFE");


  const [
    lastMessage,
    setLastMessage
  ] = useState(
    "Application ready"
  );


  const trackingRef =
    useRef<ReturnType<
      typeof setInterval
    > | null>(null);


  /*
  ============================================================
  LOAD SAVED TOURIST ID
  ============================================================
  */

  useEffect(() => {

    loadTouristId();

    checkServer();

    return () => {

      if (
        trackingRef.current
      ) {

        clearInterval(
          trackingRef.current
        );

      }

    };

  }, []);


  /*
  ============================================================
  SAVE TOURIST ID
  ============================================================
  */

  async function saveTouristId(
    value: string
  ) {

    try {

      await AsyncStorage.setItem(
        "tourist_id",
        value
      );

    }

    catch (error) {

      console.log(
        "Unable to save tourist ID",
        error
      );

    }

  }


  /*
  ============================================================
  LOAD TOURIST ID
  ============================================================
  */

  async function loadTouristId() {

    try {

      const saved =
        await AsyncStorage.getItem(
          "tourist_id"
        );

      if (saved) {

        setTouristId(
          saved
        );

      }

    }

    catch (error) {

      console.log(
        "Unable to load tourist ID",
        error
      );

    }

  }


  /*
  ============================================================
  CHECK SERVER
  ============================================================
  */

  async function checkServer() {

    try {

      setServerStatus(
        "Checking..."
      );

      const response =
        await fetch(
          BACKEND_URL +
          "/health"
        );


      if (
        !response.ok
      ) {

        throw new Error(
          "Server unavailable"
        );

      }


      const data =
        await response.json();


      if (
        data.status ===
        "healthy"
      ) {

        setServerStatus(
          "ONLINE"
        );

      }

      else {

        setServerStatus(
          "CONNECTED"
        );

      }


      setLastMessage(
        "Backend connected"
      );

    }

    catch (error) {

      setServerStatus(
        "OFFLINE"
      );

      setLastMessage(
        "Cannot connect to backend"
      );

    }

  }


  /*
  ============================================================
  REQUEST LOCATION PERMISSION
  ============================================================
  */

  async function requestLocationPermission() {

    const {
      status
    } =
      await Location.requestForegroundPermissionsAsync();


    if (
      status !== "granted"
    ) {

      Alert.alert(
        "Location Permission",
        "Location permission is required for tourist safety monitoring."
      );

      return false;

    }


    return true;

  }


  /*
  ============================================================
  GET CURRENT LOCATION
  ============================================================
  */

  async function getCurrentLocation() {

    const permission =
      await requestLocationPermission();


    if (!permission) {

      return null;

    }


    try {

      const result =
        await Location.getCurrentPositionAsync(
          {
            accuracy:
              Location.Accuracy.High,
          }
        );


      const newLocation = {

        latitude:
          result.coords.latitude,

        longitude:
          result.coords.longitude,

        accuracy:
          result.coords.accuracy,
      };


      setLocation(
        newLocation
      );


      return newLocation;

    }

    catch (error) {

      Alert.alert(
        "Location Error",
        "Unable to obtain current GPS location."
      );

      return null;

    }

  }


  /*
  ============================================================
  SEND LOCATION TO SERVER
  ============================================================
  */

  async function sendLocation(
    coordinates: Coordinates
  ) {

    try {

      const response =
        await fetch(
          BACKEND_URL +
          "/location",
          {
            method: "POST",

            headers: {
              "Content-Type":
                "application/json",
            },

            body: JSON.stringify(
              {
                tourist_id:
                  touristId,

                latitude:
                  coordinates.latitude,

                longitude:
                  coordinates.longitude,

                accuracy:
                  location.accuracy,

                status:
                  status,
              }
            ),
          }
        );


      if (
        !response.ok
      ) {

        throw new Error(
          "Location request failed"
        );

      }


      const data =
        await response.json();


      if (
        data.geofence
      ) {

        setStatus(
          data.geofence.status
        );

      }


      setLastMessage(
        "Location sent successfully"
      );

    }

    catch (error) {

      setLastMessage(
        "Location upload failed"
      );

      console.log(
        "Location upload error",
        error
      );

    }

  }


  /*
  ============================================================
  UPDATE LOCATION
  ============================================================
  */

  async function updateLocation() {

    const current =
      await getCurrentLocation();


    if (
      current
    ) {

      await sendLocation(
        current
      );

    }

  }


  /*
  ============================================================
  START LIVE TRACKING
  ============================================================
  */

  async function startTracking() {

    if (
      tracking
    ) {

      return;

    }


    const permission =
      await requestLocationPermission();


    if (!permission) {

      return;

    }


    setTracking(
      true
    );


    setLastMessage(
      "Live tracking started"
    );


    await updateLocation();


    trackingRef.current =
      setInterval(
        async () => {

          await updateLocation();

        },
        5000
      );

  }


  /*
  ============================================================
  STOP LIVE TRACKING
  ============================================================
  */

  function stopTracking() {

    if (
      trackingRef.current
    ) {

      clearInterval(
        trackingRef.current
      );

      trackingRef.current =
        null;

    }


    setTracking(
      false
    );


    setLastMessage(
      "Live tracking stopped"
    );

  }


  /*
  ============================================================
  SEND SOS
  ============================================================
  */

  async function sendSOS() {

    const current =
      await getCurrentLocation();


    if (
      !current
    ) {

      Alert.alert(
        "SOS Error",
        "Current GPS location could not be obtained."
      );

      return;

    }


    try {

      const response =
        await fetch(
          BACKEND_URL +
          "/sos",
          {
            method: "POST",

            headers: {
              "Content-Type":
                "application/json",
            },

            body: JSON.stringify(
              {
                tourist_id:
                  touristId,

                latitude:
                  current.latitude,

                longitude:
                  current.longitude,

                accuracy:
                  current.accuracy,

                message:
                  "Emergency SOS activated from tourist mobile application",
              }
            ),
          }
        );


      if (
        !response.ok
      ) {

        throw new Error(
          "SOS request failed"
        );

      }


      const data =
        await response.json();


      setStatus(
        "SOS"
      );


      setLastMessage(
        "SOS sent to authority"
      );


      Alert.alert(
        "SOS SENT",
        "Emergency alert has been sent to the authority dashboard."
      );


      console.log(
        "SOS response:",
        data
      );

    }

    catch (error) {

      Alert.alert(
        "SOS Failed",
        "Unable to send emergency alert to the backend."
      );

      console.log(
        "SOS error:",
        error
      );

    }

  }


  /*
  ============================================================
  CONFIRM SOS
  ============================================================
  */

  function confirmSOS() {

    Alert.alert(
      "Emergency SOS",
      "Are you sure you want to send an emergency alert?",
      [
        {
          text: "Cancel",
          style: "cancel",
        },

        {
          text: "SEND SOS",
          style: "destructive",
          onPress: sendSOS,
        },
      ]
    );

  }

  SAVE TOURIST ID
  ============================================================

    value: string

    setTouristId(
      value
    );

    await saveTouristId(
      value
    );

  }
